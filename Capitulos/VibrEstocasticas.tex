\chapter{Vibraciones Estocásticas}

\section{Proceso Estocástico}
Una \underline{variable aleatoria} $X$ es una variable que contiene un grado de incertidumbre. Se caracteriza por una distribución (ej. Distribución normal o lognormal). Una ``realización'' de una variable aleatoria toma un único valor. \\

Ejemplos de variables aleatorias:
\begin{itemize}
    \item Período de retorno (``ocurrencia'') de un terremoto % $ > M_w8.5$
    \item Módulo de Young de un material (Hormigón: $E_c$)
    \item Resistencia característica (Acero: $F_y$)
    \item Geometría
    \item Intensidad de un terremoto en un sitio específico.
\end{itemize}

Un \underline{proceso estocástico} (p.e.)  $\mathbb{X}$ es una colección de variables aleatorias para cada $t \in T$. Para los casos prácticos de este curso, la indexación $t$ puede ser el tiempo, por lo que la colección es secuencial. Además, un p.e. puede ser continuo (imposible de hacer en computador) o discreto (conjunto, arreglo o vector).
\begin{align*} 
    \Longrightarrow \mathbb{X} = \{X(t): t \in T\} =\{X(t_1), X(t_2), ..., X(t_m)\}
\end{align*}

Una realización de un proceso estocástico corresponde a la realización de todas las variables aleatorias de su colección. \\

\section{Caracterización de Proceso Estocástico}
\par Existen 2 formas de describir un p.e. $\mathbb{X} = \{X(t)$, $t\in T\}$:

\begin{itemize}
    \item \textbf{Función de distribución conjunta}\\
    La función de distribución conjunta de un proceso estocástico se define como la probabilidad conjunta de que el proceso tome ciertos valores en diferentes momentos del tiempo. Matemáticamente, se expresa como:
    \begin{align*}
         F_{\mathbb{X}}(\tend{x}) &= \mathbb{P}[\mathbb{X} \leq \tend{x}]\\
        \Longrightarrow F_{\mathbb{X}}(x_1,x_2,...,x_m) &= \mathbb{P}[X_1\leq x_1 \cap X_2 \leq x_2 \cap \hdots \cap X_m \leq x_m]
    \end{align*}
    
    Si el p.e es visto como una variable aleatoria que varía en función del tiempo:
    \begin{align*}
        \mathbb{P}[X(t_1)\leq x_1 \cap X(t_2)\leq x_2 \cap ... \cap X(t_m) \leq x_m ] = F_{\mathbb{X}}(x_1,x_2,...,x_m;t_1,t_2,...,t_m) 
    \end{align*}
    
    Esta expresión nos proporciona la probabilidad de que el proceso estocástico $\mathbb{X}$ en los tiempos $t_i$ tome valores que no excedan $x_i$.
    
    \item \textbf{Momentos estocásticos}\\ 
    \underline{Recordatorio}: \\
    El valor esperado de una variable aleatoria $\mathbb{E}\{X\}$ es la integral del producto entre los valores que puede tomar esa variable aleatoria $x$ y la función de distribución de probabilidad evaluada en ese valor $f_X(x) = \mathbb{P}(X=x)$.
    \begin{align*}
        \mathbb{E}\{X\} = \int_{-\infty}^{\infty} {x\cdot\mathbb{P}(X=x)dx}
    \end{align*}
    En el caso discreto:
    \begin{align*}
        \mathbb{E}\{X\} = \sum_{i=1}^{\infty}{x_i\cdot\mathbb{P}(X=x)} 
    \end{align*}

    En MATLAB, se puede ocupar \texttt{mean(X)} donde \texttt{X} es un vector de números sampleados con la distribución de su correspondiente variable aleatoria $X$. \\ 
    
    Los momentos estadísticos de 1er y 2do orden son los siguientes:
    \begin{itemize}
        \item Media $(m \times 1)$: 
        \begin{align*}
            \mu_{\mathbb{X}}(t_i)=\mathbb{E}\{X(t_i)\}
        \end{align*}
    
        \item Varianza $(m \times 1)$:
        \begin{align*}
            \sigma_{\mathbb{X}}^2(t_i)=\mathbb{E}\{[X(t_i)-\mu_X(t_i)]^2\}
        \end{align*}
    
        \item Función de autocorrelación $(m \times m)$: 
        \begin{align*}
            R_{\mathbb{X}\mathbb{X}}(t_i,t_j)=\mathbb{E}\{X(t_i)X(t_j)\}
        \end{align*}
    
        \item Función de auto-covarianza $(m \times m)$: 
        \begin{align*}
            \Gamma_{\mathbb{X}\mathbb{X}}(t_i,t_j)&=\mathbb{E}\{[X(t_i)-\mu_X(t_i)][X(t_j)-\mu_X(t_j)]\} \\
            \Gamma_{\mathbb{X}\mathbb{X}}(t_i,t_j)&=R_{\mathbb{X}\mathbb{X}}(t_i,t_j)-\mu_\mathbb{X}(t_i)\mu_\mathbb{X}(t_j)
        \end{align*}
    
        \item Coeficiente de correlación $(m \times m)$: 
        \begin{align*}
            \rho_{\mathbb{X}\mathbb{X}}(t_i,t_j) = \frac{\Gamma_{\mathbb{X}\mathbb{X}}(t_i,t_j)}{\sigma_\mathbb{X}(t_i)\sigma_\mathbb{X}(t_j)}
        \end{align*}

        Notar que: $\rho_{\mathbb{X}\mathbb{X}}(t_i, t_i) = 1$.\\
        
        Interpretación de la correlación: ``medida de independencia lineal entre variables aleatorias''.
        
        \begin{figure}[H]
            \centering
            \subfigure[]{\includegraphics[scale=0.55]{Figuras/rho1.png}}
            \subfigure[]{\includegraphics[scale=0.55]{Figuras/rho0.png}}
        \end{figure}
        \begin{itemize}
            \item $|\rho_{\mathbb{X}\mathbb{X}}(t_i,t_j)| \leq 1$
            \item Si $\rho_{\mathbb{X}\mathbb{X}}(t_i,t_j)=0$, $\forall t_i \neq t_j$, entonces el p.e. es un proceso descorrelacionado (por ejemplo: ruido blanco).
        \end{itemize}
    \end{itemize}
    En caso de dos procesos estocásticos, $\mathbb{X} = \{X(t)$, $t \in T_X\}$ e $ \mathbb{Y} = \{Y(t)$, $t \in T_Y\}$.
        \begin{itemize}
            \item Correlación cruzada: 
            \begin{align*}
                R_{\mathbb{X}\mathbb{Y}}(t_i,t_j)=\mathbb{E}\{X(t_i)Y(t_j)\}
            \end{align*}
            \item Covarianza cruzada:
            \begin{align*}
                \Gamma_{\mathbb{X}\mathbb{Y}}(t_i,t_j) &=\mathbb{E}\{ [X(t_i)-\mu_x(t_i)][Y(t_j)-\mu_y(t_j)]\} \\
                \Gamma_{\mathbb{X}\mathbb{Y}}(t_i,t_j) &= R_{\mathbb{X}\mathbb{Y}}(t_i, t_j)-\mu_{\mathbb{X}}(t_i)\mu_{\mathbb{Y}}(t_j)
            \end{align*}
            \item Coeficiente de correlación cruzado: 
            \begin{align*}
                \rho_{\mathbb{X}\mathbb{Y}}(t_i,t_j)=\frac{\Gamma_{\mathbb{X}\mathbb{Y}}(t_i,t_j)}{\sigma_{\mathbb{X}}(t_i)\sigma_{\mathbb{Y}}(t_j)}
            \end{align*}
        \end{itemize}
\end{itemize}

\section{Clasificación de P.E.}
\subsection{Estacionariedad}
\begin{enumerate}
    \item \underline{P.E. no estacionario}: La distribución de probabilidad del p.e. depende del tiempo. Las variables aleatorias de la colección no tienen la misma distribución.
    \begin{figure}[H]
    \centering
    \includegraphics[scale=0.7]{Figuras/penoesatacionario.png}
    \end{figure}
    
    \item \underline{P.E. Estrictamente estacionario}: La distribución de probabilidad del p.e. no cambia con un retraso en el tiempo:
    \begin{equation*}
        F_{\mathbb{X}}(x_1,...,x_m; t_1,...,t_m) = F_{\mathbb{X}}(x_1,...,x_m; \textcolor{brown}{t_1 + \tau,...,t_m + \tau})
    \end{equation*}
    \begin{figure}[H]
    \centering
    \includegraphics[scale=0.7]{Figuras/peestricto.png}
    \end{figure}
    \item \underline{P.E. Débilmente estacionario}: Estadísticos de 2do orden son finitos y no varían en el tiempo
    \begin{align*}
        &\phantom{\Longrightarrow}\mu_{\mathbb{X}}(t_i)=\mu_{\mathbb{X}}(t_j) = cste, \quad \sigma_{\mathbb{X}}^2(t_i)=\sigma_{\mathbb{X}}^2(t_j)=cste\\
        &\Longrightarrow R_{\mathbb{X}\mathbb{X}}(t_i,t_j) = R_{\mathbb{X}\mathbb{X}}(t_j-t_i)
    \end{align*}
\end{enumerate}

\subsection{Ergocidad}
Sea $\mathbb{X}=\{X(t): t\in T\}$ un p.e. estacionario, se dice que es \underline{ergódico} si se pueden obtener los momentos estadísticos del conjunto desde una única realización del p.e.
\begin{align*}
    \overline{\mu_\mathbb{X}(t)}&=\mu_\mathbb{X}(t) \\ 
    \\
    \overline{\mathbb{E}\{\mathbb{X}^2(t)\}} &=\mathbb{E}\{X^2(t)\} \\
    \\
    \overline{R_\mathbb{X}(\tau)} &= R_\mathbb{X}(\tau)
\end{align*}

\underline{Nota}: Un Ruido blanco es un p.e. Gaussiano, Ergódico.

\section{P.E. Gaussiano}
Un p.e. $\mathbb{X} = \{X(t)$, $t \in T\}$, se llama p.e. Gaussiano si, para cada $t\in T$, las $m$ variables aleatorias $X(t_1)$, $X(t_2)$, $...,$ $X(t_m)$ distribuyen normal.
\begin{align*}
    X_i \sim N\big(\mu_\mathbb{X}(t), \sigma_\mathbb{X}(t)\big)\\
    \Longrightarrow \mathbb{X} = \{X(t), t \in T\}\sim N\big(\mu_\mathbb{X},\Gamma_{\mathbb{X}\mathbb{X}}\big)
\end{align*}
donde $\mu_\mathbb{X}$ es el vector de las medias de las variables aleatorias del p.e. y $\Gamma_{\mathbb{X}\mathbb{X}}$ es la matriz de auto-covarianza. \\

\underline{Notas}:
\begin{itemize}
    \item Un p.e. Gaussiano y débilmente estacionario es estrictamente estacionario.
    \item Un p.e. Gaussiano preserva las transformaciones lineales, y esto es extremadamente útil en los problemas de sistemas lineales que se verán más adelante.
\end{itemize}

\section{Función de Autocorrelación}
\underline{Definición}: Función de Autocorrelación.\\

Sea $\mathbb{X} = \{X(t), t \in T\}$ un w.s.s (wide sense stationary). La matriz de autocorrelación es constante para mismas diferencias temporales, es decir, que la función de autocorrelación sólo depende de la diferencia de tiempo.
\begin{align*}
    R_{\mathbb{X}\mathbb{X}}(t_i,t_j) = cste = R_\mathbb{X}(t_i - t_j = \tau)
\end{align*}

Por lo tanto, se define la \underline{Función de Autocorrelación} $R_\mathbb{X}(\tau)$ como:
\begin{align*}
    R_\mathbb{X}(\tau)=\mathbb{E}\{X(t)X(t+\tau)\}
\end{align*}

\underline{Características de $R_\mathbb{X}(\tau)$}:
\begin{enumerate}
    \item Es una función par $R_\mathbb{X}(\tau) = R_\mathbb{X}(-\tau)$
    \item Es finita
    \item $\left|R_\mathbb{X}(\tau)\right| \leq R_\mathbb{X}(0)$
    \item $\Longrightarrow R_\mathbb{X}(0)=\mathbb{E}\{X^2(t)\}=\sigma^2_\mathbb{X}(t)+\mu_\mathbb{X}^2(t)$
    En $\tau=0$, $R_\mathbb{X}(0)$ entrega el máximo valor de la media cuadrada del p.e. $\mathbb{X} = \{X(t), t \in T\}$.
    \item $R_\mathbb{X}(\tau)$ es una función definida positiva, es decir, para cualquier función $g(t)$:
    \begin{align*}
        g(t_1)R_\mathbb{X}(t_1-t_2)g(t_2)\geq 0
    \end{align*}
    Por lo tanto, $R_\mathbb{X}(\tau)$ es definida positiva siempre.
    \item $R_\mathbb{X}(\tau)$ es continua en $\tau$ si el p.e. es continuo.
\end{enumerate}

\section{Densidad de Poder Espectral (power spectral density, PSD)}
\underline{Definición}: Densidad de poder espectral (PSD) (Teorema de Wiener-Khintchine).\\
\par Sea $\mathbb{X}=\{X(t): t\in T\}$ un p.e. w.s.s., con función de autocorrelación $R_\mathbb{X}(\tau)$. La función de densidad de poder espectral se define como la \emph{transformada de fourier} de la \emph{función de autocorrelación}:
\begin{align*}
    S_\mathbb{X}(\omega) &=\mathcal{F}\{R_\mathbb{X}(\tau)\}
\end{align*}
Equivalentemente, se puede obtener $R_\mathbb{X}(\tau)$ desde $S_\mathbb{X}(\omega)$
\begin{align*}
    R_\mathbb{X}(\tau) &=\mathcal{F}^{-1}\{S_\mathbb{X}(\omega)\}
\end{align*}
\par \underline{Propiedades PSD}:
\begin{enumerate}
    \item Función real
    \item Función simétrica
    \item Definida positiva
    \item Área bajo la curva de $S_\mathbb{X}(\omega)$ es igual a $\mathbb{E}\{X^2(t)\} = R_\mathbb{X}(\tau=0) = \sigma^2_\mathbb{X}(t) + \mu^2_\mathbb{X}(t)$ 
\end{enumerate}

\section{Ruido blanco}
Sea $\mathbb{X}=\{X(t): t\in T\}$ un p.e. w.s.s. Gaussiano, con $\mu_\mathbb{X}(t)=0$ en todo $t \in T$, $\mathbb{X}$ es un ruido blanco si su PSD es constante:
\begin{align*}
    S_X(\omega) &= S_0, \quad \forall \omega \in (-\infty,+\infty)\\
    \Longrightarrow R_X(\tau) &= 2\pi S_0\delta (\tau)
\end{align*}

donde $R_\mathbb{X}(\tau)$ es el delta de Dirac.

\subsection{Ruido blanco de banda limitada (band-limited white noise, BLWN)}
Sea $\mathbb{X}$ un p.e., w.s.s., es un \emph{ruido blanco de banda limitada} (BLWN) si:
\begin{align*}
    S_\mathbb{X}(\omega) & = \left\{
\begin{array}{ll}
      S_0 & |\omega|\leq \omega_c \\
      0 & |\omega|> \omega_c \\
\end{array} 
\right. \\
    R_\mathbb{X}(\tau) & = R_\mathbb{X}(0)\frac{\sin(\omega_c \tau)}{\omega_c \tau}\\
    & = R_\mathbb{X}(0)\text{sinc}(\omega_c \tau)
\end{align*}

\end{figure}

\section{Respuesta estocástica de sistemas lineales}
Sea un sistema LTI:
\begin{align*}
    \tend{\dot{x}} =\tenq{A}\tend{x} + \tenq{B}u(t), \quad \tend{x}(0)=\tend{0}
\end{align*}

y asumiendo que $u(t)$ es una realización de un p.e. $\mathbb{U} = \{U(t), t \in T\}$ w.s.s. con $\mu_\mathbb{U}(t) = 0$. \\

\par \underline{Objetivo}: Determinar la respuesta $\tend{x}(t)$, la cual se puede analizar como un p.e. w.s.s., $\mathbb{X} = \{X(t)$, $t \in T\}$ con $\mu_\mathbb{X}(t)=0$\\

La derivada temporal de la autocovarianza $\tenq{\Gamma}_\mathbb{X}(t)$ dado que $\mathbb{X}(t)=0$:
\begin{align*}
    \dot{\Gamma}_\mathbb{X}(t)&=\frac{d}{dt}\mathbb{E}\{XX^\top\}=\mathbb{E}\{\frac{d}{dt}(XX^\top)\}\\
    &= \mathbb{E}\{\dot{X}X^\top+X\dot{X}^\top\}\\
    &= \mathbb{E}\{(AX+BU)X^\top+X(AX+BU)^\top\}\\
    &= A\mathbb{E}\{XX^\top\}+B\mathbb{E}\{UX^\top\}+\mathbb{E}\{XX^\top\}A^\top+\mathbb{E}\{XU^\top\}B^\top
\end{align*}

Definiendo $Q=\mathbb{E}\{BUX^\top\}+\mathbb{E}\{XU^\top B^\top\}$
\begin{align*}
    \dot{\Gamma}_\mathbb{X} &= A\Gamma_\mathbb{X}+\Gamma_\mathbb{X}A^\top+Q \quad \text{Ecuación diferencial de Lyapunov}
\end{align*}

Si $\tend{u}(t)$ es un ruido blanco, es decir, $\mu_\mathbb{U}=0$, $\mathbb{E}\{U(t)U(t+\tau)\}=2\pi s_o\delta (\tau)$, con $S_\mathbb{U}(\omega)=S_0$ y recordndo que la sol analítica de $\tend{x}(t)$ es:

\begin{equation*}
    x(t)=\phi(t)x_0+\int_0^t \phi(t-\tau)Bu(\tau)d\tau
\end{equation*}

Desarrollemos el primer término de $Q$ de la ecuación diferencial de Lyapunov:
\begin{align*}
    \mathbb{E}\{BUX^\top \} &=  \mathbb{E}\{BU(t)[\int_0^t \phi(t-\tau)Bu(\tau)d\tau]^\top\}\\
    &= B\int_0^t \mathbb{E}\{U(t)[U^\top (t) B^\top \phi^\top (t-\tau)] \} d\tau\\
    &= B\int_0^t \mathbb{E}\{U(t)U^\top (t)\} B^\top \phi^\top (t-\tau) d\tau
\end{align*}

Dado que $U$ es un ruido blanco con $\mathbb{E}\{U(t)U(t+\tau)\} = 2\pi s_0 \delta (\tau)$. Se puede demostrar que:
\begin{align*}
    \mathbb{E}\{BUX^\top \}=\frac{1}{2}B2\phi s_0B^\top
\end{align*}

Por lo tanto:
\begin{align*}
    Q &= \mathbb{E}\{BUX^\top \}+\mathbb{E}\{BU^\top X^\top \}=2\mathbb{E}\{BUX^\top \}\\
    Q &= B(2\pi s_0)B^\top
\end{align*}

Para resolver $\Gamma_\mathbb{X}(t)$, se calcula $Q$ que depende de $s_0$ y $B$:
\begin{align*}
    \dot{\Gamma}_\mathbb{X}(t)=A\Gamma_\mathbb{X}+\Gamma_\mathbb{X}A^\top +Q
\end{align*}

La solución es única ssi A es Hurwitz.\\
\par \underline{Notas}:
\begin{itemize}
    \item $Q$ es una matriz simétrica, definida positiva. Entonces, $s_0$ es una matriz simétrica, definida positiva.
    \item $\Gamma_X$ es una matriz simétrica, definida positiva.
\end{itemize}

\subsection{Autocovarianza caso estacionario}
Estudiando el caso de la respuesta en estado estacionario $t\to \infty$
\begin{align*}
    \lim_{t \to \infty} \dot{\Gamma}_\mathbb{X}(t)=0
\end{align*}

En este caso, la Ecuación de Lyapunov queda de la siguiente forma:
\begin{align*}
   0=A\Gamma_\mathbb{X}+\Gamma_\mathbb{X}A^\top +Q
\end{align*}

Donde ahora $\Gamma_\mathbb{X}$ es la autocovarianza en estado estacionario de los estados $\mathbb{X}$.\\

\subsection{Autocovarianza de la respuesta}
La autocovarianza de la respuesta $\Gamma_\mathbb{Y}$, considerando:
\begin{align*}
    y(t) = Cx(t)
\end{align*}

donde $y(t)$ es un p.e., w.s.s., $\mathbb{Y} = \{Y(t), t \in T\}$ con $\mu_\mathbb{Y}(t)=0$.\\

La matriz de autocovarianza de la respuesta se determina de la siguiente manera:
\begin{align*}
    \Gamma_\mathbb{Y}=\mathbb{E}\{YY^\top\} &= \mathbb{E}\{(CX)(CX)^\top\}\\
     &= \mathbb{E}\{CXX^\top C^\top\}\\
      &= C \mathbb{E}\{XX^\top\}C^\top\\
       &= C \Gamma_\mathbb{X} C^\top\\
\end{align*}

Donde $\Gamma_\mathbb{X}$ es la solución de la ecuación de Lyapunov en estado estacionario.

\subsection{Error Cuadrático Medio RMS}
Sea el proceso estocástico $\mathbb{X} = \{X(t), t \in T\}$, su Error Cuadrático Medio (Root mean square error, RMS) es una función que cuantifica... y se calcula como:

\begin{align*}
    RMS_\mathbb{X}(t) = \sqrt{\mathbb{E}\{(X(t) - \mu_\mathbb{X}(t))^2\}}
\end{align*}

Cuando $\mu_\mathbb{X}(t) = 0$
\begin{align*}
    RMS_\mathbb{X}(t) = \sqrt{\mathbb{E}\{X(t)^2\}} = \sqrt{\mathbb{E}\{XX^\top\}} \\
    \Longrightarrow RMS_\mathbb{X}(t) = \sqrt{\Gamma_{\mathbb{X}}(t)} 
\end{align*}


\section{Modelos Estocásticos en Ingeniería Estructural}
Los parámetros de modelos de sistemas dinámico contienen incertidumbre que requiere ser incorporada para un mejor entendimiento estadístico del problema. Algunos parámetros se pueden considerar como variables aleatorias. Otros, como las vibraciones, pueden ser modeladas como procesos estocásticos. Por ejemplo.

    \item Viento.
    \begin{itemize}
        \item Daurenport
        \item Kaimal-Vokerman
    \end{itemize}
    \item Tráfico (vehicular, peatonal)
    \item Rugosidad pavimento
    \item Oleaje.
    \begin{itemize}
        \item Pierson-Moskowitz
    \end{itemize}
\end{itemize}

\subsection{Simulación de registros sintéticos - Modelo Kanai-Tajimi (K-T)}
El modelo K-T utiliza un p.e. Gaussiano estacionario escalado en función del tiempo para representar la aceleración del suelo producto de un evento sísmico. \\

Consideremos el siguiente suelo.

\begin{figure}[H]
\centering
\includegraphics[scale=0.85]{Figuras/KT.png}
\end{figure}
El suelo es modelado como un sistema SDOF donde la aceleración total del oscilador es lo que representa la aceleración del suelo. En este caso, la EDM del oscilador SDOF es la siguiente ($a$ aceleración relativa del suelo respecto a la roca)
\begin{align*}
    \ddot{a}+2\xi_g\omega_g\dot{a}+\omega_g^2a= W(t) 
\end{align*}

Considerando la aceleración de la roca base $W(t)$ como p.e. $\mathbb{W} = \{W(t), t \in T\}$ ruido blanco, es \begin{align*}
    \mu_\mathbb{W}(t) = 0 \\
    R_\mathbb{W}(\tau) = 2\pi S_0 \delta(\tau) \\ S_\mathbb{W}(\omega) = S_0 
\end{align*}

Donde $S_0$ es el poder espectral del sismo en roca y está relacionado con una medida de intensidad del registro sísmico, ej: PGA o Sa(T).\\

La aceleración total del suelo es  $\ddot{u}_g = \ddot{a} + W(t)$ (en campo abierto):
\begin{align*}
    \ddot{u}_g = -2\xi_g\omega_g \dot{a}-\omega_g^2a
\end{align*}

En representación espacio-estado:
\begin{align*}
    \tend{\dot{x}}_g= \tenq{A}_g\tend{x}_g+\tenq{B}_gW(t) \\
    \tend{y}_g = \tenq{C}_g\tend{x}_g
\end{align*}

Donde
\begin{align*}
    \tend{x_g} = \begin{Bmatrix}
        a \\
        \dot{a}
    \end{Bmatrix} \\
    \tend{y} = \begin{Bmatrix}
        \ddot{u}_g
    \end{Bmatrix} \\
    \tenq{A}_g = \begin{bmatrix}
        0 & 1 \\
        -\omega_g^2  & -2\xi_g\omega_g
    \end{bmatrix} \\
    \tenq{B}_g = \begin{bmatrix}
        0 \\
        1
    \end{bmatrix} \\
    \tenq{C}_g = \begin{bmatrix}
        -\omega_g^2 & -2\xi_g\omega_g
    \end{bmatrix} \\
    \tenq{D}_g = \tenq{0}
\end{align*}

Se puede demostrar que la función de respuesta en frecuencia (FRF) del estado de suelo es:

\begin{align*}
    H_{KT}(\Omega) &= \frac{\omega_g^2+i2\xi_g\omega_g\Omega}{(\omega_g^2-\Omega^2)+i2\xi_g\omega_g\Omega}\\
    |H_{KT}(\Omega)|^2 &= \frac{\omega_g^4+4\xi_g^2\omega_g^2\Omega^2}{(\omega_g^2-\Omega^2)^2+4\xi_g^2\omega_g^2\Omega^2}
\end{align*}

\begin{figure}[H]

\subsubsection{Filtro K-T aumentado}
Equivalentemente, se puede generar un sistema aumentado (Filtro K-T + Estructura) que tome como input el ruido blanco $W(t)$ y retorne como output la respuesta de la estructura $\tend{y}(t)$.

\begin{figure}[H]
\centering
\includegraphics[scale=0.6]{Figuras/estado_estacionario_KT.png}
\end{figure}

El sistema aumentado en representación espacio-estado $x_a = \begin{Bmatrix}
        x_s \\
        x_g
    \end{Bmatrix}\\$
\begin{align*}
    \dot{x}_a=A_ax_a+B_aW(t)\\
    y = C_ax_a \\
    A_a = \begin{bmatrix}
        A_s & B_s C_g \\
        0 & A_g
    \end{bmatrix}; 
    B_a = \begin{bmatrix}
        0 \\
        B_g
    \end{bmatrix}; 
    C_a = \begin{bmatrix}
        C_s & 0
    \end{bmatrix}
\end{align*}

Sea $\Gamma_{\mathbb{X}_a}$ la autocovarianza de los estados del sistema aumentado. Se puede obtener la autocovarianza de la respuesta $\Gamma_\mathbb{Y}$ desde la ecuación de Lyapunov en estado estacionario:
\begin{align*}
    0 &= A_a\Gamma_{\mathbb{X}_a} +\Gamma_{\mathbb{X}_a}A_a^\top+Q_a\\
    \Gamma_\mathbb{Y} &= C_a\Gamma_{\mathbb{X}_a}C_a^\top
\end{align*}

\subsubsection{Modulación}
El registro sísmico sintético que tenemos hasta el momento $\ddot{u}_g$ es un ruido blanco estacionario. Pero, en realidad los registros sísmicos son p.e. no estacionarios. Por lo tanto, hay que escalarlo con una \emph{función de modulación} $\psi(t)$ para que le de "forma" de un registro sísmico.

\end{figure}

\centering
\includegraphics[scale=0.6]{Figuras/func_mod_KT2.png}
\end{figure}

Notar que se mantiene $\mu_{\mathbb{U}_g}(t) = 0$, pero, $\sigma_{\mathbb{U}_g}(t) \neq 0$. \\

Ahora el sistema aumentado tiene matrices que varían en el tiempo.
\begin{align*}
    \dot{x}_a(t) &= A_a(t)x_a(t)+B_a(t)Z(t)\\
    y(t) &= C_a(t)x_a(t)
\end{align*}

La covarianza en este caso se debe resolver integrando la ecuación diferencial de Lyapunov donde $\Gamma_{\mathbb{X}_a}(t)$ está en función del tiempo ($\dot{\Gamma}_{\mathbb{X}_a} \neq 0$):
\begin{align*}
    \dot{\Gamma}_{\mathbb{X}_a}(t) = A_a(t)\Gamma_{\mathbb{X}_a}(t)+\Gamma_{\mathbb{X}_a}(t)A_a^\top (t))+Q_a(t)
\end{align*}

Como $\Gamma_{\mathbb{X}_a}(t)$ varía en el tiempo, entonces $ \Gamma_\mathbb{Y}(t)$ también varía en el tiempo.
