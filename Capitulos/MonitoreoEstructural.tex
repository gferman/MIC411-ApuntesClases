\chapter{Técnicas de Monitoreo Estructural}

\section{Introducción}

Nunca se está seguro de las propiedades \emph{exactas} de una estructura. 

\begin{enumerate}
    \item Obras civiles ``envejecen'':
    \begin{itemize}
        \item Daño/deterioro estructura $\longrightarrow$ Cambio en las propiedades de materiales
    \end{itemize}
    \item Incertidumbre:
    \begin{itemize}
        \item Propiedades nominales utilizadas en diseño
        \item Los modelos nominales no necesariamente representan fielmente el comportamiento real de la estructura
    \end{itemize}
\end{enumerate}

Un buen proyecto de monitoreo estructural (Structural health monitoring, SHM) tiene múltiples objetivos más allá de la instrumentación.

\begin{figure}[H]
    \centering
    \includegraphics[scale=0.5]{Figuras/sysID.png}
\end{figure}

Para el monitoreo estructural se busca:

\begin{itemize}
    \item Determinar un modelo matemático de un sistema.
    \item Realizar predecicción: desempeño, daño,
\end{itemize}

\section{Identificación de Sistemas}

Dado un sistema LTI representado por su FRF $G_{YU}(\omega)$, buscamos determinar $G_{YU}(\omega)$ dado $u(t)$ y $y(t)$ conocidos.\\

\begin{figure}[H]
    \centering
    \includegraphics[scale=0.6]{Figuras/UY.png}
\end{figure}

\begin{align*}
    Y(\omega) &= G_{YU}(\omega)U(\omega) \quad \text{(FRF)}\\
    y(t) &= (g_{YU} \ast u)(t) \quad \text{(FRI)} \\
    G_{YU}(\omega) &= \mathcal{F}\{g_{YU}(t)\}
\end{align*}

\subsection{Análisis Modal Experimental}

\par Asumir que $u(t)$ e $y(t)$ son procesos estocásticos $\mathbb{U}$ e $\mathbb{Y}$ estacionarios Gaussianos de media cero. \\

\par Recordando el teorema de Wiener-Khinchin:

\begin{align*}
    S_{\mathbb{X}\mathbb{X}}(\omega) &= \mathcal{F}\{R_{\mathbb{X}\mathbb{X}}(\tau)\}\\
    R_{\mathbb{X}\mathbb{X}}(\tau) &= \mathcal{F}^{-1}\{S_{\mathbb{X}\mathbb{X}}(\omega)\}
\end{align*}

donde $R_{\mathbb{X}\mathbb{X}}(\tau) =\mathbb{E}\{X(t)X(t+\tau)\}$ para  un p.e. estacionario.\\

Se puede demostrar que: 

\begin{align*}
    S_{\mathbb{X}\mathbb{X}}(\omega) = \mathbb{E}\{X(\omega)X^\ast(\omega)\} \quad \text{Poder espectral}
\end{align*}

donde $X(\omega)=\mathcal{F}\{x(t)\}$ y $()^\ast$: transpuesta conjugada.\\

\par Un primer alcance:

\begin{align*}
    S_{\mathbb{Y}\mathbb{Y}}(\omega) &= \mathbb{E}\{Y(\omega)Y^\ast (\omega)\}\\
    &= \mathbb{E}\Bigl\{\bigl(G_{YU}(\omega)U(\omega)\bigr)\bigl(G_{YU}(\omega)U(\omega)\bigr)^\ast\Bigr\}\\
    &= \mathbb{E}\Bigl\{G_{YU}(\omega)U(\omega)U^\ast(\omega)G^\ast_{YU}(\omega) \Bigr\}
\end{align*}

Si el sistema es SISO, y es determinístico:

\begin{align*}
    S_{\mathbb{Y}\mathbb{Y}}(\omega) &= G_{YU}(\omega)G^\ast_{YU}(\omega)\mathbb{E}\Bigl\{U(\omega)U^\ast(\omega) \Bigr\}\\
    S_{\mathbb{Y}\mathbb{Y}}(\omega) &= |G_{YU}(\omega)|^2 S_{\mathbb{U}\mathbb{U}}(\omega)\\
    \tag{*}
    |G_{YU}(\omega)|^2 &= \frac{S_{\mathbb{Y}\mathbb{Y}}(\omega)}{S_{\mathbb{Y}\mathbb{Y}}(\omega)}
\end{align*}

\underline{Nota}: $|G_{YU}(\omega)|^2$ es una función real y nos entrega sólo información de la magnitud del sistema y no de la fase. Por lo tanto, \underline{este método queda descartado}.

\subsection{Método \texorpdfstring{$H_1$}{H1}}

\begin{align*}
    Y(\omega) &= G_{YU}(\omega)U(\omega) \quad / \cdot U^*(\omega)\\
    Y(\omega)U^\ast(\omega) &= G_{YU}(\omega)U(\omega)U^\ast(\omega) \quad / \mathbb{E}\{\cdot\}\\
    \mathbb{E}\{Y(\omega)U^\ast(\omega)\} &= G_{YU}(\omega)\mathbb{E}\{U(\omega)U^\ast(\omega)\} \\
    S_{\mathbb{Y}\mathbb{U}}(\omega) &= G_{YU}(\omega)S_{\mathbb{U}\mathbb{U}}(\omega)\\
    \hat{G}_{YU}(\omega) &= \frac{S_{\mathbb{Y}\mathbb{U}}(\omega)}{S_{UU}(\omega)} \quad \text{Estimador $H_1$}
\end{align*}

donde $\mathbb{E}\{Y(\omega)U^\ast(\omega)\} =S_{\mathbb{Y}\mathbb{U}}(\omega)$ es el PSD cruzado.

Notar que este método tiene tanto magnitud:  $|\hat{G}_{YU}(\omega)|$, como fase: $\angle G_{YU}(\omega)$.

\subsection{Método \texorpdfstring{$H_2$}{H2}}

\begin{align*}
    Y(\omega) &= G_{YU}(\omega)U(\omega)\\
    S_{\mathbb{Y}\mathbb{Y}} &= G_{YU}(\omega)S_{\mathbb{Y}\mathbb{U}}(\omega)\\
    \hat{G}_{YU}(\omega) &= \frac{S_{\mathbb{Y}\mathbb{Y}}(\omega)}{S_{\mathbb{Y}\mathbb{U}}(\omega)} \quad \text{Estimador $H_2$}
\end{align*}

En ausencia de ruido en $u(t)$ y $y(t)$, ambos estimadores $H_1$ y $H_2$ coinciden (caso ideal).\\

Situación con ruido:\\

\begin{figure}[H]
    \centering
    \includegraphics[scale=0.7]{Figuras/H1_ruido.png}

\par \underline{Nota}: 

\begin{itemize}
    \item Ruido $m(t)$ y $n(t)$ son p.e. estacionarios no correlacionados $\Longrightarrow R_{\mathbb{M}\mathbb{U}} = \mathbb{E}\{m(t) u(t+\tau)\}=0$
    \item Ruido $m(t)$ y $u(t)$ descorr. $\Longrightarrow \mathbb{E}\{m(t) u(t+\tau)\}=0$
    \item Ruido $n(t)$ y $y(t)$ descorr. $\Longrightarrow \mathbb{E}\{n(t) y(t+\tau)\}=0$
\end{itemize}

\begin{align*}
    X(\omega) &= U(\omega)+M(\omega)\\
    X(\omega)X^\ast(\omega) &= \bigl[ U(\omega)+M(\omega)\bigr]\bigl[ U(\omega)+M(\omega)\bigr]^\ast\\
    X(\omega)X^\ast(\omega)&= U(\omega)U^\ast(\omega)+M(\omega)^\ast (\omega)+M(\omega)U^\ast(\omega)+U(\omega)M^\ast(\omega) \quad / \mathbb{E}\{\cdot\}\\
    S_{\mathbb{X}\mathbb{X}}(\omega) &= S_{\mathbb{U}\mathbb{U}}(\omega)+S_{\mathbb{M}\mathbb{M}}(\omega)+S_{\mathbb{M}\mathbb{U}}(\omega)+S_{\mathbb{U}\mathbb{M}}(\omega)\\
    \tag{1}
    S_{\mathbb{X}\mathbb{X}}(\omega) &= S_{\mathbb{U}\mathbb{U}}(\omega)+S_{\mathbb{M}\mathbb{M}}(\omega)
\end{align*}

\begin{align*}
    Z(\omega) &= Y(\omega)+N(\omega)\\
    \tag{2}
    S_{\mathbb{Z}\mathbb{Z}}(\omega) &= S_{\mathbb{Y}\mathbb{Y}}(\omega)+S_{\mathbb{N}\mathbb{N}}(\omega)
\end{align*}

Estimador $H_1$: 

\begin{align*}
    Z(\omega) &= G_{ZX}(\omega)(\omega)\\
    Z(\omega)X^\ast(\omega) &= G_{ZX}(\omega)X(\omega)X^\ast(\omega)\\
    \tag{3}
    S_{\mathbb{Z}\mathbb{X}}(\omega) &= G_{\mathbb{Z}\mathbb{X}}(\omega)S_{\mathbb{X}\mathbb{X}}(\omega)
\end{align*}

\begin{align*}
    Z(\omega)X^\ast(\omega)&= \bigl[ Y(\omega)+N(\omega) \bigr] \bigl[ U(\omega)+M(\omega)\bigr]^\ast\\
    Z(\omega)X^\ast(\omega)&= Y(\omega)U\ast(\omega)+Y(\omega)M^\ast(\omega)+N(\omega)U^\ast(\omega)+N(\omega)M(\omega)^\ast \quad / \mathbb{E}\{\cdot\}\\
    S_{\mathbb{Z}\mathbb{X}}(\omega) &= S_{\mathbb{Y}\mathbb{U}}(\omega)+S_{\mathbb{Y}\mathbb{M}}(\omega)+S_{\mathbb{N}\mathbb{U}}(\omega)+S_{\mathbb{N}\mathbb{M}}(\omega)\\
    \tag{4}
    S_{\mathbb{Z}\mathbb{X}}(\omega) &= S_{\mathbb{Y}\mathbb{U}}(\omega)
\end{align*}

Entonces, de Ec. (3):

\begin{align*}
    G_{ZX}(\omega) &= \frac{S_{\mathbb{Z}\mathbb{X}}(\omega)}{S_{\mathbb{X}\mathbb{X}}(\omega)}=\frac{S_{\mathbb{Y}\mathbb{U}}(\omega)}{S_{\mathbb{U}\mathbb{U}}(\omega)+S_{\mathbb{M}\mathbb{M}}(\omega)}\\
    &= \frac{S_{\mathbb{Y}\mathbb{U}}(\omega)/S_{\mathbb{U}\mathbb{U}}(\omega)}{1+S_{\mathbb{M}\mathbb{M}}(\omega)/S_{\mathbb{U}\mathbb{U}}(\omega)}\\
    H_1 &= \frac{S_{\mathbb{Z}\mathbb{X}}(\omega)}{S_{\mathbb{X}\mathbb{X}}(\omega)} = \frac{G_{YU}(\omega)}{1+S_{\mathbb{M}\mathbb{M}}(\omega)/S_{\mathbb{U}\mathbb{U}}(\omega)}
\end{align*}

donde $S_{\mathbb{M}\mathbb{M}}(\omega)/S_{\mathbb{U}\mathbb{U}}(\omega)$ es efecto del ruido de entrada $M(\omega)$.\\

\par Luego, $H_1$ es un limite inferior de $G_{YU}(\omega)$.\\

\par De manera similar:

\begin{align*}
    H_2 &= \frac{S_{\mathbb{Z}\mathbb{Z}}(\omega)}{S_{\mathbb{Z}\mathbb{X}}(\omega)} = G_{YU}(\omega)+\frac{S_{\mathbb{N}\mathbb{N}}(\omega)}{S_{\mathbb{Y}\mathbb{Y}}(\omega)}
\end{align*}

donde $S_{\mathbb{N}\mathbb{N}}(\omega)/S_{\mathbb{Y}\mathbb{Y}}(\omega)$ es efecto del ruido de salida.\\

\par Luego, $H_2$ es un límite superior de $G_{YU}(\omega)$

\subsection{Función de coherencia}

\begin{align*}
    \gamma_{ZX}^2(\omega) = \frac{|S_{\mathbb{Z}\mathbb{X}}(\omega)|^2}{S_{\mathbb{Z}\mathbb{Z}}(\omega)S_{\mathbb{X}\mathbb{X}}(\omega)}
\end{align*}

Si $\gamma_{ZX}^2=1$, entonces existe coherencia entre entrada y salida $\Longrightarrow$ $H_1$ y $H_2$ son buenos estimadores.\\
